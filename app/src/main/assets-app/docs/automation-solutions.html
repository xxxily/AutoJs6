App 启动与等待

适用场景: 打开目标 App, 等待首页或关键控件稳定后继续执行。

依赖能力: accessibility. 风险等级: medium.

常见失败原因: 应用名称不匹配或未安装; 首页文案随语言、版本、登录状态变化; 无障碍服务未启用。

"auto";

var APP_NAME = "目标应用";
var READY_TEXT = "首页";

if (!app.launchApp(APP_NAME)) {
    throw new Error("无法启动应用: " + APP_NAME);
}

var ready = auto.waitUntil(text(READY_TEXT), {
    timeout: 12000,
    interval: 500,
    stableCount: 2,
    captureOnFailure: true
});

if (!ready.ok) {
    throw new Error("首页未就绪: " + ready.reason);
}

toastLog("已进入首页: " + APP_NAME);

可复制模板: 替换 APP_NAME 和 READY_TEXT, 保留 auto.waitUntil() 的稳定等待和失败诊断。

列表滚动查找

适用场景: 在长列表中滚动查找目标文本或控件, 找到后稳定点击。

依赖能力: accessibility. 风险等级: medium.

常见失败原因: 列表容器不可滚动; 目标文本被截断或动态加载; 滚动次数过少。

"auto";

var KEYWORD = "目标文本";
var result = auto.findWithScroll(textContains(KEYWORD), null, {
    timeout: 15000,
    maxScrolls: 12,
    direction: "down",
    captureOnFailure: true
});

if (!result.ok) {
    throw new Error("未找到列表项: " + result.reason);
}

var clicked = auto.stableClick(result.uiObject, { timeout: 3000 });
if (!clicked.ok) {
    throw new Error("列表项点击失败: " + clicked.reason);
}

可复制模板: 替换 KEYWORD, 必要时把第二个参数换成可滚动容器选择器。

表单填写

适用场景: 等待输入框和按钮稳定, 依次填写字段并提交。

依赖能力: accessibility. 风险等级: medium.

常见失败原因: 输入框 id 变化; 键盘遮挡提交按钮; 提交前需要额外勾选协议或处理验证码。

"auto";

var account = auto.stableSetText(id("account"), "demo@example.com", { timeout: 8000 });
if (!account.ok) throw new Error("账号输入失败: " + account.reason);

var password = auto.stableSetText(id("password"), "change-me", { timeout: 8000 });
if (!password.ok) throw new Error("密码输入失败: " + password.reason);

var submit = auto.stableClick(text("登录"), { timeout: 8000, stableCount: 2 });
if (!submit.ok) throw new Error("提交失败: " + submit.reason);

可复制模板: 将 id("account")、id("password") 和提交按钮文案替换为目标页面的稳定选择器。

OCR 文字点击

适用场景: 无障碍节点没有文本时, 使用屏幕感知/OCR 查找文字并点击。

依赖能力: accessibility, screen_capture, ocr. 风险等级: medium.

常见失败原因: 截图权限未授权; OCR 引擎不可用; 文字被遮挡、置信度过低或命中区域不可点击。

"auto";

var target = vision.findText("确定", {
    sources: "a11y+ocr",
    timeout: 6000,
    minConfidence: 0.35
});

if (!target) {
    throw new Error("未通过 OCR/无障碍找到目标文字");
}

if (target.selector) {
    var clicked = auto.stableClick(target.selector, { timeout: 3000 });
    if (!clicked.ok) throw new Error("selector 点击失败: " + clicked.reason);
} else {
    var action = target.suggestedAction;
    click(action.x, action.y);
}

可复制模板: 优先使用 target.selector 做稳定点击, 仅在没有选择器时使用 OCR 坐标兜底。

截图找图

适用场景: 请求截图权限后, 用模板图在当前屏幕中查找并点击。

依赖能力: screen_capture, storage. 风险等级: medium.

常见失败原因: 模板图片路径不存在; 目标缩放、主题或分辨率变化; 截图权限未授权或会话失效。

if (!requestScreenCapture()) {
    throw new Error("截图权限请求失败");
}

var template = images.read("./assets/button.png");
if (!template) throw new Error("模板图片不存在");

try {
    var screen = captureScreen();
    var point = images.findImage(screen, template, { threshold: 0.82 });
    if (!point) throw new Error("未找到模板图");
    click(point.x, point.y);
} finally {
    template.recycle();
}

project.json 提示: 声明 capabilities: ["screen_capture", "storage"] 和 assets: ["assets/button.png"]。

定时任务

适用场景: 创建带重试、互斥和超时策略的定时任务, 并查询最近运行记录。

依赖能力: background_run, exact_alarm. 风险等级: medium.

常见失败原因: 系统限制精确闹钟或后台启动; 脚本路径不存在; 任务因互斥跳过; 重启后未允许开机启动。

var task = tasks.addDailyTask({
    path: files.join(files.cwd(), "daily-sync.js"),
    time: "08:30",
    maxRetries: 2,
    retryBackoffMillis: 60 * 1000,
    mutex: "daily-sync",
    timeoutMillis: 5 * 60 * 1000
});

log("task id: " + task.id);
log("queue: " + JSON.stringify(tasks.queryTimedTaskQueue()));
log("runs: " + JSON.stringify(tasks.queryTimedTaskRuns({ taskId: task.id, limit: 5 })));

project.json 提示: 声明 capabilities: ["background_run", "exact_alarm"]。

Shizuku 应用管理

适用场景: 使用结构化 Shizuku API 查询、强停、授权或清理应用, 避免手写 pm/am 命令。

依赖能力: shizuku. 风险等级: high.

常见失败原因: Shizuku 服务未运行或未授权 AutoJs6; 目标包名错误; 授权权限不属于目标应用 manifest。

"auto";

var PACKAGE = "com.example.target";
var state = shizuku.state();
if (!state.available) {
    throw new Error("Shizuku 不可用: " + state.reason);
}

var info = shizuku.package.info(PACKAGE);
log("target: " + JSON.stringify(info));

var stopped = shizuku.app.forceStop(PACKAGE);
log("forceStop: " + JSON.stringify(stopped));

project.json 提示: 声明 capabilities: ["shizuku"], privilegedPolicy.shizuku = true, 并将高风险策略设为提示或拒绝。

插件 OCR

适用场景: 通过 Paddle OCR 插件能力识别图片文字, 并在打包项目中声明插件依赖。

依赖能力: ocr, screen_capture. 风险等级: medium.

常见失败原因: Paddle OCR 插件未安装、未启用或版本不兼容; 打包项目未声明 pluginDependencies; 图片路径不存在。

console.show();

var img = images.read("./test.png");
if (!img) throw new Error("图片不存在: ./test.png");

try {
    var results = ocr.paddle.detect(img, { useSlim: true, cpuThreadNum: 4 });
    Array.from(results).forEach(function (item) {
        log(item.label + " " + item.confidence + " " + JSON.stringify(item.bounds));
    });
} finally {
    img.recycle();
}

project.json 提示: 声明 capabilities: ["ocr", "screen_capture"] 和 pluginDependencies: ["org.autojs.autojs.plugin.paddle.ocr"]。
